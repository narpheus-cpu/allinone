function mtInitSlots() { const grid = document.getElementById('mtSlotsGrid'); if(!grid) return; grid.innerHTML = ''; for (let i = 1; i <= 10; i++) { let rawData = localStorage.getItem('word_slot_' + i); let div = document.createElement('div'); if (rawData) { let parsedData = JSON.parse(rawData); let slotData = Array.isArray(parsedData) ? { name: "슬롯 " + i, words: parsedData } : parsedData; // ★ 표시된 카드 수 계산 let starredCount = slotData.words ? slotData.words.filter(w => w.starred).length : 0; div.className = 'mt-slot has-data'; div.innerHTML = `
${slotData.name}
카드 수: ${slotData.words.length}개
★표기카드수: ${starredCount}개
`; div.onclick = () => mtPromptAction(i, slotData); } else { div.className = 'mt-slot'; div.innerHTML = `
슬롯 ${i}
[비어 있음]
클릭하여 파일 추가
`; div.onclick = () => mtTriggerUpload(i); } grid.appendChild(div); } }
